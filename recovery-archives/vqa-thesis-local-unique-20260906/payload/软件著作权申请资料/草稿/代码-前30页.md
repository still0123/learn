async function api(path, options = {}) {
  const response = await fetch(path, {
    headers: { "Content-Type": "application/json" },
    ...options,
  });
  const data = await response.json();
  if (!response.ok) {
    throw new Error(data.error || "请求失败");
  }
  return data;
}
function escapeHtml(value) {
  return String(value ?? "")
    .replaceAll("&", "&amp;")
    .replaceAll("<", "&lt;")
    .replaceAll(">", "&gt;")
    .replaceAll('"', "&quot;");
}
function tagsHtml(tags = []) {
  if (!tags.length) return "<span class=\"tag\">无标签</span>";
  return `<div class="tag-row">${tags.map((tag) => `<span class="tag">${escapeHtml(tag)}</span>`).join("")}</div>`;
}
function semanticTagsHtml(tags = []) {
  if (!tags.length) return "<span class=\"tag semantic\">无语义要素</span>";
  return `<div class="tag-row">${tags.map((tag) => `<span class="tag semantic">${escapeHtml(tag)}</span>`).join("")}</div>`;
}

async function withBusy(buttonSelector, task) {
  const button = $(buttonSelector);
  const original = button ? button.textContent : "";
  if (button) {
    button.disabled = true;
    button.textContent = "处理中";
  }
  try {
    return await task();
  } finally {
    if (button) {
      button.disabled = false;
      button.textContent = original;
    }
  }
}
function switchView(name) {
  $all(".nav-item").forEach((item) => item.classList.toggle("active", item.dataset.view === name));
  $all(".view").forEach((item) => item.classList.toggle("active", item.id === `view-${name}`));
  $("#view-title").textContent = viewTitles[name] || "总览";
}
async function loadState() {
  const data = await api("/api/state");
  state.materials = data.materials || [];
  state.history = data.history || [];
  state.evidencePackages = data.evidence_packages || [];
  state.metrics = data.metrics || {};

async function submitMaterial(event) {
  event.preventDefault();
  await withBusy("#material-form .primary-button", async () => {
    const form = new FormData(event.currentTarget);
    const payload = Object.fromEntries(form.entries());
    const data = await api("/api/materials", {
      method: "POST",
      body: JSON.stringify(payload),
    });
    state.selectedMaterialId = data.material.material_id;
    await loadState();
    switchView("materials");
  });
}

async function deleteMaterial(materialId) {
  const material = state.materials.find((item) => item.material_id === materialId);
  if (!confirm(`确认删除素材“${material?.title || materialId}”？相关证据包和问答历史也会删除。`)) return;
  await api(`/api/materials/${encodeURIComponent(materialId)}`, { method: "DELETE" });
  if (state.selectedMaterialId === materialId) {
    state.selectedMaterialId = "";
  }
  await loadState();
}
